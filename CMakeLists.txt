cmake_minimum_required(VERSION 3.25)
project(UmbrellaShell VERSION 0.2.0 LANGUAGES CXX)

option(UMBRELLA_BUILD_APP "Build Windows settings window" ON)
include(CTest)
add_library(umbrella_core INTERFACE)
target_include_directories(umbrella_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(umbrella_core INTERFACE cxx_std_20)

if(UMBRELLA_BUILD_APP)
    if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The host requires Windows x64 and MSVC (Qt msvc2022_64)")
    endif()
    enable_language(RC)
    # The installer uses a flat Windows runtime layout.
    # Qt's deploy API must use the same bindir, otherwise DLLs go into bin/.
    set(CMAKE_INSTALL_BINDIR ".")
    include(GNUInstallDirs)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Core Qml Quick)
    qt_standard_project_setup(REQUIRES 6.8)
    add_library(UmbrellaTaskbarModule SHARED src/platform/windows/explorer/TaskbarModule.cpp)
    add_executable(UmbrellaTaskbarHost src/platform/windows/explorer/TaskbarModuleHost.cpp)
    add_library(UmbrellaMenuModule SHARED src/platform/windows/explorer/TaskbarMenuModule.cpp)
    add_executable(UmbrellaMenuHost WIN32 src/platform/windows/explorer/TaskbarMenuHost.cpp)
    set_target_properties(UmbrellaTaskbarModule UmbrellaMenuModule PROPERTIES PREFIX "")
    # Hook DLLs execute inside Explorer, outside the app-local CRT search context.
    # Keep their runtime private; no CRT objects cross the Win32 protocol boundary.
    set_target_properties(UmbrellaTaskbarModule UmbrellaMenuModule PROPERTIES
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(UmbrellaMenuModule PRIVATE /bigobj)
    foreach(target UmbrellaTaskbarModule UmbrellaTaskbarHost UmbrellaMenuModule UmbrellaMenuHost)
        target_compile_features(${target} PRIVATE cxx_std_20)
        # Keep HRESULT diagnostics without embedding local source/home paths.
        target_compile_definitions(${target} PRIVATE WINRT_NO_SOURCE_LOCATION UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /sdl)
        target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT)
        # On the tested Explorer, CFG-instrumented taskbar hooks fail the
        # cross-process handshake (the in-process fixture still passes).
        if(NOT target STREQUAL "UmbrellaTaskbarModule")
            target_compile_options(${target} PRIVATE /guard:cf)
            target_link_options(${target} PRIVATE /guard:cf)
        endif()
        target_link_libraries(${target} PRIVATE user32)
    endforeach()
    target_link_libraries(UmbrellaTaskbarModule PRIVATE comctl32 advapi32 shell32 gdi32 windowsapp d2d1)
    target_link_libraries(UmbrellaTaskbarHost PRIVATE advapi32 ole32 oleaut32 uiautomationcore gdi32 shell32)
    target_link_libraries(UmbrellaMenuModule PRIVATE comctl32 windowsapp)
    target_link_libraries(UmbrellaMenuHost PRIVATE shell32)
    add_library(umbrella_platform STATIC src/platform/windows/DockPreview.cpp src/platform/windows/DockPreview.h src/platform/windows/DockWindows.cpp src/platform/windows/DockWindows.h src/platform/windows/WindowBackdrop.cpp src/platform/windows/SettingsWindowControl.cpp src/platform/windows/TaskbarRuntime.cpp src/platform/windows/TaskbarState.cpp src/platform/windows/TaskbarAppearance.cpp src/platform/windows/TaskbarModuleSession.cpp)
    target_include_directories(umbrella_platform PUBLIC src)
    target_compile_features(umbrella_platform PUBLIC cxx_std_20)
    target_link_libraries(umbrella_platform PRIVATE Qt6::Gui dwmapi shell32 user32 gdi32 advapi32 ole32 propsys windowsapp)
    target_compile_definitions(umbrella_platform PRIVATE WINRT_NO_SOURCE_LOCATION UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
    # Qt Quick owns the single rendering stack through its D3D11 RHI.
    add_library(umbrella_render INTERFACE)
    target_link_libraries(umbrella_render INTERFACE d3d11 dxgi)
    add_library(umbrella_presentation INTERFACE)
    target_link_libraries(umbrella_presentation INTERFACE Qt6::Qml Qt6::Quick umbrella_render)
    configure_file(cmake/Version.h.in generated/Version.h @ONLY)
    configure_file(packaging/windows/version.rc.in generated/version.rc @ONLY)
    qt_add_executable(UmbrellaShell src/app/main.cpp src/app/DockController.cpp src/app/DockController.h src/app/TaskbarSettings.cpp src/app/TaskbarSettings.h ${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc)
    set_target_properties(UmbrellaShell PROPERTIES WIN32_EXECUTABLE TRUE)
    target_include_directories(UmbrellaShell PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_link_libraries(UmbrellaShell PRIVATE Qt6::Core umbrella_core umbrella_platform umbrella_presentation)
    target_compile_definitions(UmbrellaShell PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_source_files_properties(src/presentation/Settings.qml PROPERTIES QT_RESOURCE_ALIAS Settings.qml)
    set_source_files_properties(src/presentation/TaskbarPage.qml PROPERTIES QT_RESOURCE_ALIAS TaskbarPage.qml)
    set_source_files_properties(src/presentation/TaskbarConfiguration.qml PROPERTIES QT_RESOURCE_ALIAS TaskbarConfiguration.qml)
    set_source_files_properties(src/presentation/TaskbarLayout.js PROPERTIES QT_RESOURCE_ALIAS TaskbarLayout.js)
    set_source_files_properties(src/presentation/Dock.qml PROPERTIES QT_RESOURCE_ALIAS Dock.qml)
    set_source_files_properties(src/presentation/DockPage.qml PROPERTIES QT_RESOURCE_ALIAS DockPage.qml)
    qt_add_qml_module(UmbrellaShell URI Umbrella VERSION 1.0 QML_FILES src/presentation/Dock.qml src/presentation/DockPage.qml src/presentation/Settings.qml src/presentation/TaskbarPage.qml src/presentation/TaskbarConfiguration.qml src/presentation/TaskbarLayout.js)
    qt_add_resources(UmbrellaShell branding PREFIX "/" FILES assets/umbrella.png)
    foreach(target UmbrellaShell umbrella_platform)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /sdl /guard:cf)
    endforeach()
    target_link_options(UmbrellaShell PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
    install(TARGETS UmbrellaShell RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(TARGETS UmbrellaTaskbarModule UmbrellaTaskbarHost UmbrellaMenuModule UmbrellaMenuHost RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    # App-local MSVC runtime avoids an elevated VC redistributable installer.
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ${CMAKE_INSTALL_BINDIR})
    include(InstallRequiredSystemLibraries)
    # Include QML imports in the standalone Windows deployment.
    qt_generate_deploy_qml_app_script(TARGET UmbrellaShell OUTPUT_SCRIPT deploy_script
        DEPLOY_TOOL_OPTIONS --no-opengl-sw)
    install(SCRIPT ${deploy_script})
    install(FILES README.md CHANGELOG.md DESTINATION .)
    install(DIRECTORY docs/ DESTINATION docs FILES_MATCHING PATTERN "*.md"
        PATTERN "licenses" EXCLUDE PATTERN "AGENTS*.md" EXCLUDE)
    install(DIRECTORY docs/licenses/ DESTINATION docs/licenses
        PATTERN "AGENTS*.md" EXCLUDE PATTERN ".agents" EXCLUDE PATTERN ".codex" EXCLUDE)
    if(BUILD_TESTING)
        find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
        add_executable(umbrella_snapshot_protocol_tests tests/taskbar_snapshot_protocol_tests.cpp)
        target_link_libraries(umbrella_snapshot_protocol_tests PRIVATE umbrella_core)
        add_test(NAME taskbar_snapshot_protocol COMMAND umbrella_snapshot_protocol_tests)
        add_executable(umbrella_dock_thumbnail_tests tests/dock_thumbnail_tests.cpp)
        target_link_libraries(umbrella_dock_thumbnail_tests PRIVATE umbrella_core dwmapi)
        target_compile_definitions(umbrella_dock_thumbnail_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        add_test(NAME dock_thumbnail COMMAND umbrella_dock_thumbnail_tests)
        add_executable(umbrella_dock_fixture_tests tests/dock_fixture_tests.cpp src/app/DockController.cpp src/app/DockController.h)
        target_link_libraries(umbrella_dock_fixture_tests PRIVATE umbrella_platform umbrella_presentation psapi)
        target_compile_options(umbrella_dock_fixture_tests PRIVATE /utf-8)
        target_compile_definitions(umbrella_dock_fixture_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        qt_add_resources(umbrella_dock_fixture_tests dock_fixture_qml PREFIX "/qt/qml/Umbrella"
            FILES src/presentation/Dock.qml src/presentation/DockPage.qml)
        qt_add_resources(umbrella_dock_fixture_tests dock_fixture_branding PREFIX "/" FILES assets/umbrella.png)
        add_test(NAME dock_fixture COMMAND umbrella_dock_fixture_tests)
        set_tests_properties(dock_fixture PROPERTIES TIMEOUT 30)
        # Explicit private-process XAML experiment; no Explorer or global input.
        add_executable(umbrella_xaml_layout_tests EXCLUDE_FROM_ALL tests/taskbar_xaml_layout_tests.cpp)
        target_link_libraries(umbrella_xaml_layout_tests PRIVATE umbrella_core windowsapp user32 shell32)
        target_compile_definitions(umbrella_xaml_layout_tests PRIVATE WINRT_NO_SOURCE_LOCATION UNICODE _UNICODE NOMINMAX)
        target_compile_options(umbrella_xaml_layout_tests PRIVATE /utf-8)
        target_link_options(umbrella_xaml_layout_tests PRIVATE /MANIFEST:NO)
        add_custom_command(TARGET umbrella_xaml_layout_tests POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                ${CMAKE_CURRENT_SOURCE_DIR}/tests/taskbar_xaml_layout_tests.manifest
                $<TARGET_FILE:umbrella_xaml_layout_tests>.manifest VERBATIM)
        add_executable(umbrella_session_tests tests/taskbar_session_tests.cpp)
        target_link_libraries(umbrella_session_tests PRIVATE Qt6::Core umbrella_platform)
        target_compile_options(umbrella_session_tests PRIVATE /utf-8)
        add_test(NAME taskbar_session COMMAND umbrella_session_tests)
        set_tests_properties(taskbar_session PROPERTIES TIMEOUT 15)
        add_executable(umbrella_recovery_guard_tests
            tests/taskbar_recovery_guard_tests.cpp src/app/TaskbarSettings.cpp src/app/TaskbarSettings.h)
        target_link_libraries(umbrella_recovery_guard_tests PRIVATE Qt6::Gui umbrella_platform umbrella_presentation)
        target_compile_definitions(umbrella_recovery_guard_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_compile_options(umbrella_recovery_guard_tests PRIVATE /utf-8)
        add_test(NAME taskbar_recovery_guard COMMAND umbrella_recovery_guard_tests)
        add_executable(umbrella_resume_desktop_tests EXCLUDE_FROM_ALL
            tests/resume_desktop_tests.cpp src/app/TaskbarSettings.cpp src/app/TaskbarSettings.h)
        target_link_libraries(umbrella_resume_desktop_tests PRIVATE Qt6::Test umbrella_platform umbrella_presentation)
        target_compile_definitions(umbrella_resume_desktop_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_compile_options(umbrella_resume_desktop_tests PRIVATE /utf-8)
        # Explicit desktop-only target; never part of unattended CTest or the default build.
        add_executable(umbrella_configuration_desktop_test EXCLUDE_FROM_ALL
            tests/taskbar_configuration_desktop_test.cpp src/app/TaskbarSettings.cpp src/app/TaskbarSettings.h)
        target_link_libraries(umbrella_configuration_desktop_test PRIVATE Qt6::Test umbrella_platform umbrella_presentation)
        target_compile_definitions(umbrella_configuration_desktop_test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_compile_options(umbrella_configuration_desktop_test PRIVATE /utf-8)
        target_link_libraries(umbrella_configuration_desktop_test PRIVATE ole32 oleaut32 uiautomationcore user32)
        add_executable(umbrella_menu_patch_tests tests/taskbar_menu_patch_tests.cpp)
        target_include_directories(umbrella_menu_patch_tests PRIVATE src)
        target_compile_features(umbrella_menu_patch_tests PRIVATE cxx_std_20)
        target_compile_definitions(umbrella_menu_patch_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        add_test(NAME taskbar_menu_patch COMMAND umbrella_menu_patch_tests)
        add_executable(umbrella_menu_settings_fixture WIN32 tests/menu_settings_fixture.cpp)
        target_include_directories(umbrella_menu_settings_fixture PRIVATE src)
        target_compile_features(umbrella_menu_settings_fixture PRIVATE cxx_std_20)
        target_compile_options(umbrella_menu_settings_fixture PRIVATE /utf-8)
        target_link_libraries(umbrella_menu_settings_fixture PRIVATE shell32 user32)
        add_executable(umbrella_settings_lifetime_tests tests/settings_window_lifetime_tests.cpp)
        target_include_directories(umbrella_settings_lifetime_tests PRIVATE src)
        target_compile_features(umbrella_settings_lifetime_tests PRIVATE cxx_std_20)
        target_link_libraries(umbrella_settings_lifetime_tests PRIVATE Qt6::Gui)
        add_test(NAME settings_window_lifetime COMMAND umbrella_settings_lifetime_tests)
        set_tests_properties(settings_window_lifetime PROPERTIES TIMEOUT 10)
        find_program(UMBRELLA_POWERSHELL NAMES pwsh powershell REQUIRED)
        add_test(NAME taskbar_menu_settings_launch COMMAND ${UMBRELLA_POWERSHELL} -NoProfile -ExecutionPolicy Bypass
            -File ${CMAKE_CURRENT_SOURCE_DIR}/tests/Test-MenuSettingsLaunch.ps1
            -HostPath $<TARGET_FILE:UmbrellaMenuHost> -FixturePath $<TARGET_FILE:umbrella_menu_settings_fixture>)
        add_test(NAME taskbar_module_geometry COMMAND UmbrellaTaskbarHost --fixture)
        add_test(NAME taskbar_component_observer COMMAND UmbrellaTaskbarHost --fixture-inspect-components)
        set_tests_properties(taskbar_component_observer PROPERTIES TIMEOUT 15)
        add_test(NAME settings_ui_smoke COMMAND UmbrellaShell --ui-smoke)
        set_tests_properties(settings_ui_smoke PROPERTIES TIMEOUT 20)
        add_test(NAME host_diagnostics COMMAND UmbrellaShell --diagnostics)
        set_tests_properties(host_diagnostics PROPERTIES PASS_REGULAR_EXPRESSION "updatesEnabled.*false")
        add_test(NAME host_unknown_argument COMMAND UmbrellaShell --unknown)
        set_tests_properties(host_unknown_argument PROPERTIES WILL_FAIL TRUE)
    endif()
endif()
