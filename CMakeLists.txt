cmake_minimum_required(VERSION 3.25)
project(UmbrellaShell VERSION 0.1.0 LANGUAGES CXX)

option(UMBRELLA_BUILD_APP "Build Windows settings window" ON)
include(CTest)
add_library(umbrella_core INTERFACE)
target_include_directories(umbrella_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(umbrella_core INTERFACE cxx_std_20)

if(UMBRELLA_BUILD_APP)
    if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The host requires Windows x64 and MSVC (Qt msvc2022_64)")
    endif()
    enable_language(RC)
    # The installer uses a flat Windows runtime layout.
    # Qt's deploy API must use the same bindir, otherwise DLLs go into bin/.
    set(CMAKE_INSTALL_BINDIR ".")
    include(GNUInstallDirs)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Core Qml Quick)
    qt_standard_project_setup(REQUIRES 6.8)
    add_library(UmbrellaTaskbarModule SHARED src/platform/windows/explorer/TaskbarModule.cpp)
    add_executable(UmbrellaTaskbarHost src/platform/windows/explorer/TaskbarModuleHost.cpp)
    add_library(UmbrellaMenuModule SHARED src/platform/windows/explorer/TaskbarMenuModule.cpp)
    add_executable(UmbrellaMenuHost WIN32 src/platform/windows/explorer/TaskbarMenuHost.cpp)
    set_target_properties(UmbrellaTaskbarModule UmbrellaMenuModule PROPERTIES PREFIX "")
    target_compile_options(UmbrellaMenuModule PRIVATE /bigobj)
    foreach(target UmbrellaTaskbarModule UmbrellaTaskbarHost UmbrellaMenuModule UmbrellaMenuHost)
        target_compile_features(${target} PRIVATE cxx_std_20)
        target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /sdl /guard:cf)
        target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
        target_link_libraries(${target} PRIVATE user32)
    endforeach()
    target_link_libraries(UmbrellaTaskbarModule PRIVATE comctl32 advapi32 shell32 gdi32 windowsapp d2d1)
    target_link_libraries(UmbrellaTaskbarHost PRIVATE advapi32 ole32 oleaut32 uiautomationcore gdi32)
    target_link_libraries(UmbrellaMenuModule PRIVATE comctl32 windowsapp)
    target_link_libraries(UmbrellaMenuHost PRIVATE shell32)
    add_library(umbrella_platform STATIC src/platform/windows/WindowBackdrop.cpp src/platform/windows/SettingsWindowControl.cpp src/platform/windows/TaskbarState.cpp src/platform/windows/TaskbarAppearance.cpp src/platform/windows/TaskbarModuleSession.cpp)
    target_include_directories(umbrella_platform PUBLIC src)
    target_compile_features(umbrella_platform PUBLIC cxx_std_20)
    target_link_libraries(umbrella_platform PRIVATE Qt6::Gui dwmapi shell32 user32 gdi32 advapi32)
    target_compile_definitions(umbrella_platform PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
    # Qt Quick owns the single rendering stack through its D3D11 RHI.
    add_library(umbrella_render INTERFACE)
    target_link_libraries(umbrella_render INTERFACE d3d11 dxgi)
    add_library(umbrella_presentation INTERFACE)
    target_link_libraries(umbrella_presentation INTERFACE Qt6::Qml Qt6::Quick umbrella_render)
    configure_file(cmake/Version.h.in generated/Version.h @ONLY)
    configure_file(packaging/windows/version.rc.in generated/version.rc @ONLY)
    qt_add_executable(UmbrellaShell src/app/main.cpp src/app/TaskbarSettings.cpp src/app/TaskbarSettings.h ${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc)
    set_target_properties(UmbrellaShell PROPERTIES WIN32_EXECUTABLE TRUE)
    target_include_directories(UmbrellaShell PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_link_libraries(UmbrellaShell PRIVATE Qt6::Core umbrella_core umbrella_platform umbrella_presentation)
    target_compile_definitions(UmbrellaShell PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_source_files_properties(src/presentation/Settings.qml PROPERTIES QT_RESOURCE_ALIAS Settings.qml)
    set_source_files_properties(src/presentation/TaskbarPage.qml PROPERTIES QT_RESOURCE_ALIAS TaskbarPage.qml)
    qt_add_qml_module(UmbrellaShell URI Umbrella VERSION 1.0 QML_FILES src/presentation/Settings.qml src/presentation/TaskbarPage.qml)
    qt_add_resources(UmbrellaShell branding PREFIX "/" FILES assets/umbrella.png)
    foreach(target UmbrellaShell umbrella_platform)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /sdl /guard:cf)
    endforeach()
    target_link_options(UmbrellaShell PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
    install(TARGETS UmbrellaShell RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(TARGETS UmbrellaTaskbarModule UmbrellaTaskbarHost UmbrellaMenuModule UmbrellaMenuHost RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    # App-local MSVC runtime avoids an elevated VC redistributable installer.
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ${CMAKE_INSTALL_BINDIR})
    include(InstallRequiredSystemLibraries)
    # Include QML imports in the standalone Windows deployment.
    qt_generate_deploy_qml_app_script(TARGET UmbrellaShell OUTPUT_SCRIPT deploy_script
        DEPLOY_TOOL_OPTIONS --no-opengl-sw)
    install(SCRIPT ${deploy_script})
    install(FILES README.md CHANGELOG.md DESTINATION .)
    install(DIRECTORY docs/ DESTINATION docs FILES_MATCHING PATTERN "*.md"
        PATTERN "licenses" EXCLUDE PATTERN "AGENTS*.md" EXCLUDE)
    install(DIRECTORY docs/licenses/ DESTINATION docs/licenses
        PATTERN "AGENTS*.md" EXCLUDE PATTERN ".agents" EXCLUDE PATTERN ".codex" EXCLUDE)
    if(BUILD_TESTING)
        add_executable(umbrella_menu_patch_tests tests/taskbar_menu_patch_tests.cpp)
        target_include_directories(umbrella_menu_patch_tests PRIVATE src)
        target_compile_features(umbrella_menu_patch_tests PRIVATE cxx_std_20)
        target_compile_definitions(umbrella_menu_patch_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        add_test(NAME taskbar_menu_patch COMMAND umbrella_menu_patch_tests)
        add_executable(umbrella_menu_settings_fixture WIN32 tests/menu_settings_fixture.cpp)
        target_include_directories(umbrella_menu_settings_fixture PRIVATE src)
        target_compile_features(umbrella_menu_settings_fixture PRIVATE cxx_std_20)
        target_compile_options(umbrella_menu_settings_fixture PRIVATE /utf-8)
        target_link_libraries(umbrella_menu_settings_fixture PRIVATE shell32 user32)
        add_executable(umbrella_settings_lifetime_tests tests/settings_window_lifetime_tests.cpp)
        target_include_directories(umbrella_settings_lifetime_tests PRIVATE src)
        target_compile_features(umbrella_settings_lifetime_tests PRIVATE cxx_std_20)
        target_link_libraries(umbrella_settings_lifetime_tests PRIVATE Qt6::Gui)
        add_test(NAME settings_window_lifetime COMMAND umbrella_settings_lifetime_tests)
        set_tests_properties(settings_window_lifetime PROPERTIES TIMEOUT 10)
        find_program(UMBRELLA_POWERSHELL NAMES pwsh powershell REQUIRED)
        add_test(NAME taskbar_menu_settings_launch COMMAND ${UMBRELLA_POWERSHELL} -NoProfile -ExecutionPolicy Bypass
            -File ${CMAKE_CURRENT_SOURCE_DIR}/tests/Test-MenuSettingsLaunch.ps1
            -HostPath $<TARGET_FILE:UmbrellaMenuHost> -FixturePath $<TARGET_FILE:umbrella_menu_settings_fixture>)
        add_test(NAME taskbar_module_geometry COMMAND UmbrellaTaskbarHost --fixture)
        add_test(NAME settings_ui_smoke COMMAND UmbrellaShell --ui-smoke)
        set_tests_properties(settings_ui_smoke PROPERTIES TIMEOUT 20)
        add_test(NAME host_diagnostics COMMAND UmbrellaShell --diagnostics)
        set_tests_properties(host_diagnostics PROPERTIES PASS_REGULAR_EXPRESSION "updatesEnabled.*false")
        add_test(NAME host_unknown_argument COMMAND UmbrellaShell --unknown)
        set_tests_properties(host_unknown_argument PROPERTIES WILL_FAIL TRUE)
    endif()
endif()

if(BUILD_TESTING)
    add_executable(umbrella_policy_tests tests/update_policy_tests.cpp)
    target_link_libraries(umbrella_policy_tests PRIVATE umbrella_core)
    add_test(NAME update_policy COMMAND umbrella_policy_tests)
endif()
