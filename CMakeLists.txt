cmake_minimum_required(VERSION 3.25)
project(UmbrellaShell VERSION 0.1.0 LANGUAGES CXX)

option(UMBRELLA_BUILD_APP "Build Windows Shell Bar" ON)
include(CTest)
add_library(umbrella_core INTERFACE)
target_include_directories(umbrella_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(umbrella_core INTERFACE cxx_std_20)

if(UMBRELLA_BUILD_APP)
    if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The host requires Windows x64 and MSVC (Qt msvc2022_64)")
    endif()
    enable_language(RC)
    # The installer and recovery shortcuts use a flat Windows runtime layout.
    # Qt's deploy API must use the same bindir, otherwise DLLs go into bin/.
    set(CMAKE_INSTALL_BINDIR ".")
    include(GNUInstallDirs)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Core Qml Quick)
    qt_standard_project_setup(REQUIRES 6.8)
    add_library(umbrella_platform STATIC src/platform/windows/ExplorerRecovery.cpp src/platform/windows/DesktopBar.cpp src/platform/windows/TaskbarSession.cpp src/platform/windows/TaskbarSession.h src/platform/windows/TaskbarVisibility.cpp)
    target_include_directories(umbrella_platform PUBLIC src)
    target_compile_features(umbrella_platform PUBLIC cxx_std_20)
    target_link_libraries(umbrella_platform PUBLIC shell32 user32 advapi32 psapi Qt6::Core)
    target_compile_definitions(umbrella_platform PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    # Qt Quick owns the single rendering stack through its D3D11 RHI.
    add_library(umbrella_render INTERFACE)
    target_link_libraries(umbrella_render INTERFACE d3d11 dxgi)
    add_library(umbrella_presentation INTERFACE)
    target_link_libraries(umbrella_presentation INTERFACE Qt6::Qml Qt6::Quick umbrella_render)
    configure_file(cmake/Version.h.in generated/Version.h @ONLY)
    configure_file(packaging/windows/version.rc.in generated/version.rc @ONLY)
    qt_add_executable(UmbrellaShell src/app/main.cpp src/app/ShellController.cpp src/app/ShellController.h ${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc)
    set_target_properties(UmbrellaShell PROPERTIES WIN32_EXECUTABLE TRUE)
    target_include_directories(UmbrellaShell PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_link_libraries(UmbrellaShell PRIVATE Qt6::Core umbrella_core umbrella_platform umbrella_presentation)
    target_compile_definitions(UmbrellaShell PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_source_files_properties(src/presentation/Bar.qml PROPERTIES QT_RESOURCE_ALIAS Bar.qml)
    set_source_files_properties(src/presentation/Settings.qml PROPERTIES QT_RESOURCE_ALIAS Settings.qml)
    qt_add_qml_module(UmbrellaShell URI Umbrella VERSION 1.0 QML_FILES src/presentation/Bar.qml src/presentation/Settings.qml)
    qt_add_resources(UmbrellaShell branding PREFIX "/" FILES assets/umbrella.png)
    foreach(target UmbrellaShell umbrella_platform)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /sdl /guard:cf)
    endforeach()
    target_link_options(UmbrellaShell PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
    install(TARGETS UmbrellaShell RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    # App-local MSVC runtime avoids an elevated VC redistributable installer.
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ${CMAKE_INSTALL_BINDIR})
    include(InstallRequiredSystemLibraries)
    # Include QML imports in the standalone Windows deployment.
    qt_generate_deploy_qml_app_script(TARGET UmbrellaShell OUTPUT_SCRIPT deploy_script)
    install(SCRIPT ${deploy_script})
    install(FILES README.md docs/THIRD_PARTY.md DESTINATION docs)
    if(BUILD_TESTING)
        add_executable(umbrella_guard_tests tests/taskbar_guard_tests.cpp)
        target_link_libraries(umbrella_guard_tests PRIVATE umbrella_platform)
        target_compile_definitions(umbrella_guard_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_compile_options(umbrella_guard_tests PRIVATE /W4 /permissive- /utf-8)
        add_test(NAME taskbar_guard COMMAND umbrella_guard_tests)
        set_tests_properties(taskbar_guard PROPERTIES TIMEOUT 15)
        add_test(NAME shell_ui_smoke COMMAND UmbrellaShell --ui-smoke)
        set_tests_properties(shell_ui_smoke PROPERTIES TIMEOUT 20)
        add_test(NAME guard_invalid_arguments COMMAND UmbrellaShell --guard invalid 0 invalid invalid)
        set_tests_properties(guard_invalid_arguments PROPERTIES WILL_FAIL TRUE TIMEOUT 5)
        add_test(NAME host_diagnostics COMMAND UmbrellaShell --diagnostics)
        set_tests_properties(host_diagnostics PROPERTIES PASS_REGULAR_EXPRESSION "updatesEnabled.*false")
        add_test(NAME host_unknown_argument COMMAND UmbrellaShell --unknown)
        set_tests_properties(host_unknown_argument PROPERTIES WILL_FAIL TRUE)
    endif()
endif()

if(BUILD_TESTING)
    add_executable(umbrella_policy_tests tests/update_policy_tests.cpp)
    target_link_libraries(umbrella_policy_tests PRIVATE umbrella_core)
    add_test(NAME update_policy COMMAND umbrella_policy_tests)
endif()
